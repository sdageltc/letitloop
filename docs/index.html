The 10-Second Quickstart

Turn fragile multi-step LLM pipelines into atomic, crash-proof workflows.

from letitloop import durable, step

@durable(checkpoint_dir=".bench_wal")
def run_autonomous_pipeline(task_prompt: str):
    with step("research"):
        context = search_codebase(task_prompt)
        
    with step("implement"):
        diff = llm_coder(context)  # 💥 Process dies or SIGKILL fires here?
        apply_patch(diff)
        
    with step("verify"):
        # ⚡ Re-running resumes right here — research & implement are NEVER re-executed.
        run_tests()
